# 局域网文件传输工具依赖（Python 3.9+，Windows 推荐 3.10 ~ 3.14）
# TUS 分片上传方案：支持 100G 级大文件、断点续传、多设备并发
#
# 安装（任选其一）：
#   .venv\Scripts\pip install -r requirements.txt -i https://pypi.tuna.tsinghua.edu.cn/simple
#   .venv\Scripts\pip install -r requirements.txt -i https://pypi.tuna.tsinghua.edu.cn/simple --proxy http://127.0.0.1:7897

# ---- Web 框架与服务 ----
fastapi>=0.110.0
uvicorn>=0.29.0
httptools>=0.6.0            # 高性能 HTTP 解析器；大文件流式传输必需（Windows 有官方 wheel，无需编译）
tuspyserver>=4.5.0          # TUS 1.0 分片上传协议（断点续传 / 过期清理）

# ---- 二维码与窗口界面 ----
qrcode>=7.4.2               # 生成访问地址二维码
pillow>=10.2.0              # 二维码图片渲染，供 tkinter 窗口显示

# ---- 说明 ----
# tkinter 是 Python 标准库自带（Windows 官方安装包默认包含），不需要 pip 安装；
# 若用的是精简版 Python 导致 import tkinter 失败，可用 --no-gui 参数只跑控制台模式。
#
# ---- 可选：打包成 exe（仅 build.py 需要） ----
# pyinstaller>=6.0
